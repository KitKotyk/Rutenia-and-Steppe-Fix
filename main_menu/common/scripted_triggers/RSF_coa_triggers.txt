# Ruthenia UKR – Cultures

coa_def_UKR_kyivan_trigger = { # Kyivan Ruthenia
	scope:actor ?= {
		exists = culture
		culture = culture:ruthenian
	}
}

coa_def_UKR_severian_trigger = { # Severian Ruthenia
	scope:actor ?= {
		exists = culture
		culture = culture:severian
	}
}

coa_def_UKR_halician_trigger = { # Halician Ruthenia
	scope:actor ?= {
		OR = {
			AND = {
				exists = culture
				culture = culture:halychian_culture			
			}
			AND = {
				exists = culture
				culture = culture:rusyn_culture			
			}
		}
	}
}

coa_def_UKR_volhynian_trigger = { # Volhynian Ruthenia
	scope:actor ?= {		
		exists = culture
		culture = culture:volhynian_culture	
	}
}

coa_def_UKR_polesian_trigger = { # Polesian Ruthenia
	scope:actor ?= {
		exists = culture
		culture = culture:polesian_culture
	}
}

coa_def_UKR_belarusian_trigger = { # Belarus Ruthenia
	scope:actor ?= {
		OR = {
			AND = {
				exists = culture
				culture = culture:polatskian_culture			
			}
			AND = {
				exists = culture
				culture = culture:smolenskian_culture			
			}
		}
	}
}
