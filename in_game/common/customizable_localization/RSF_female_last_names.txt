# Ukrainian adjective
UkrGenderSuffixAdj = { 
	type = character
	log_loc_errors = no
	
	text = {
		localization_key = ukr_suffix_a_female
		trigger = {
			exists = this
			is_female = yes
		}
	}
	
	text = {
		localization_key = ukr_suffix_yi_male
		fallback = yes
	}
}
