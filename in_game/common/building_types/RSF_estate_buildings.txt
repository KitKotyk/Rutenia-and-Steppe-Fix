# ----- COSSACKS -----

# Cossack Colony in non-orthodox countries
cossacks_colony = {		
	audio_tier = 2	
	is_foreign = no
	max_levels = estate_building_single_level
	pop_type = tribesmen
	
	category = estate_category
	employment_size = estate_peasants_employment
	
	rural_settlement = yes
	town = yes
	city = no
	megalopolis = no
	
	estate = cossacks_estate
	build_time = estate_build_time

	country_potential = {		
	}
	
	location_potential = {
		any_pop = { 
			is_cossacks = yes
		}
		OR = {
			region = region:steppes_region
			area = area:left_bank_ukraine_area
			area = area:right_bank_ukraine_area
			area = area:podolia_area
		}
	}

	allow = {
		NOT = { owner ?= { religion ?= religion:orthodox } }
	}

	remove_if = {
		location = { 
			owner = { religion ?= religion:orthodox }
		}
	}

	possible_production_methods = {
		estate_building_input
	}

	construction_demand = estate_construct_building

	modifier = {
		local_cossacks_estate_power = 1		# 100% higher here.
		local_raw_material_output = 0.05
		local_pop_assimilation_speed_modifier = -0.5
		local_pop_conversion_speed_modifier = -0.5
	}	
}
